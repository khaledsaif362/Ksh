CREATE FUNCTION FN_Square(A INT)
RETURNS INT
DETERMINISTIC
BEGIN
    RETURN A * A;
END;


CREATE FUNCTION FN_Cube(A INT)
RETURNS INT
DETERMINISTIC
BEGIN
    RETURN A * A * A;
END;


CREATE FUNCTION FN_EvenOdd(A INT)
RETURNS VARCHAR(10)
DETERMINISTIC
BEGIN
    IF A % 2 = 0 THEN
        RETURN 'Even';
    ELSE
        RETURN 'Odd';
    END IF;
END;


CREATE FUNCTION FN_Add(A INT,B INT)
RETURNS INT
DETERMINISTIC
BEGIN
    RETURN A + B;
END;


CREATE FUNCTION FN_Max(A INT,B INT)
RETURNS INT
DETERMINISTIC
BEGIN
    IF A > B THEN
        RETURN A;
    ELSE
        RETURN B;
    END IF;
END;



CREATE FUNCTION FN_Min(A INT,B INT)
RETURNS INT
DETERMINISTIC
BEGIN
    IF A < B THEN
        RETURN A;
    ELSE
        RETURN B;
    END IF;
END;






CREATE FUNCTION FN_SumDigit(A INT)
RETURNS INT
DETERMINISTIC
BEGIN
    DECLARE S INT DEFAULT 0;

    WHILE A > 0 DO
        SET S = S + (A % 10);
        SET A = A DIV 10;
    END WHILE;

    RETURN S;
END;







CREATE FUNCTION FN_CountDigit(A INT)
RETURNS INT
DETERMINISTIC
BEGIN
    DECLARE C INT DEFAULT 0;

    WHILE A > 0 DO
        SET C = C + 1;
        SET A = A DIV 10;
    END WHILE;

    RETURN C;
END;









CREATE FUNCTION FN_LastDigit(A INT)
RETURNS INT
DETERMINISTIC
BEGIN
    RETURN A % 10;
END;







CREATE FUNCTION FN_LastDigit(A INT)
RETURNS INT
DETERMINISTIC
BEGIN
    RETURN A % 10;
END;




CREATE FUNCTION FN_Reverse(A INT)
RETURNS INT
DETERMINISTIC
BEGIN
    DECLARE R INT DEFAULT 0;

    WHILE A > 0 DO
        SET R = R * 10 + (A % 10);
        SET A = A DIV 10;
    END WHILE;

    RETURN R;
END;






CREATE FUNCTION FN_Palindrome(A INT)
RETURNS VARCHAR(5)
DETERMINISTIC
BEGIN
    DECLARE N INT;
    DECLARE R INT DEFAULT 0;

    SET N = A;

    WHILE A > 0 DO
        SET R = R * 10 + (A % 10);
        SET A = A DIV 10;
    END WHILE;

    IF N = R THEN
        RETURN 'YES';
    ELSE
        RETURN 'NO';
    END IF;
END;
